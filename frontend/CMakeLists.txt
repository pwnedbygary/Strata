# Apply the MIT upstream parser fix to a build-tree copy, never the dependency checkout.
file(READ "${STRATA_GGML_DIR}/common/parsers/qwen3-coder.cpp" _qwen_parser)
set(_old [=[                auto args = p.permute("tool-" + name + "-args", required_args);
                if (!optional_args.empty()) {
                    args = args + p.zero_or_more(p.choice(optional_args));
                }]=])
set(_new [=[                std::vector<common_peg_parser> all_args = required_args;
                all_args.insert(all_args.end(), optional_args.begin(), optional_args.end());
                auto args = all_args.empty() ? p.eps() : p.zero_or_more(p.choice(all_args));]=])
string(REPLACE "${_old}" "${_new}" _qwen_parser_fixed "${_qwen_parser}")
if(_qwen_parser_fixed STREQUAL _qwen_parser AND NOT _qwen_parser MATCHES "all_args.insert")
  message(FATAL_ERROR "Unknown Qwen tool parser revision: native parser patch needs review")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qwen3-coder.cpp" "${_qwen_parser_fixed}")
get_target_property(_common_sources llama-common SOURCES)
list(FILTER _common_sources EXCLUDE REGEX "/parsers/qwen3-coder.cpp$")
